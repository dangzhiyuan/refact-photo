# 贴纸模块实现

贴纸模块允许用户在图像上添加、移动、调整和旋转各种贴纸，这是图像编辑应用的常见功能。该模块包含以下核心功能：

1. 贴纸资源管理和展示
2. 添加贴纸到图像上
3. 贴纸变换操作（移动、缩放、旋转）
4. 贴纸层级管理（前移、后移、删除）

## 模块结构

贴纸模块由以下主要组件构成：

```
sticker/
├── StickerPanel.tsx            // 贴纸面板主容器
├── StickerList.tsx             // 贴纸列表组件
├── StickerCategory.tsx         // 贴纸分类组件
├── StickerItem.tsx             // 单个贴纸展示组件
├── StickerCanvas.tsx           // 贴纸渲染画布
├── StickerTransformer.tsx      // 贴纸变换控制器
└── StickerActions.tsx          // 贴纸操作工具栏
```

## 贴纸状态管理

贴纸功能需要专门的状态管理，用于跟踪贴纸的位置、大小、旋转等。

```typescript
// src/stores/stickerStore.ts
import { create } from 'zustand';
import { v4 as uuidv4 } from 'uuid';

// 贴纸分类
export enum StickerCategoryType {
  EMOJI = 0,
  DECORATION = 1,
  TEXT = 2,
  SHAPE = 3,
  CUSTOM = 4
}

// 贴纸对象
export interface Sticker {
  id: string;
  imageUri: string;
  position: { x: number; y: number };
  scale: number;
  rotation: number;
  zIndex: number;
  width: number;
  height: number;
  flipped: boolean;
}

// 贴纸资源
export interface StickerAsset {
  id: string;
  imageUri: string; 
  category: StickerCategoryType;
  name: string;
  width: number;
  height: number;
}

interface StickerState {
  // 贴纸列表
  stickers: Sticker[];
  selectedStickerId: string | null;
  
  // 贴纸操作
  addSticker: (asset: StickerAsset, position: { x: number; y: number }) => void;
  removeSticker: (id: string) => void;
  clearStickers: () => void;
  
  // 选择操作
  setSelectedStickerId: (id: string | null) => void;
  
  // 变换操作
  updateStickerPosition: (id: string, position: { x: number; y: number }) => void;
  updateStickerScale: (id: string, scale: number) => void;
  updateStickerRotation: (id: string, rotation: number) => void;
  flipSticker: (id: string) => void;
  
  // 层级操作
  bringToFront: (id: string) => void;
  sendToBack: (id: string) => void;
  
  // 当前分类
  currentCategoryType: StickerCategoryType;
  setCurrentCategoryType: (type: StickerCategoryType) => void;
}

export const useStickerStore = create<StickerState>((set, get) => ({
  // 贴纸列表
  stickers: [],
  selectedStickerId: null,
  
  // 贴纸操作
  addSticker: (asset, position) => set(state => {
    // 找到当前最高的z-index
    const maxZIndex = state.stickers.reduce(
      (max, sticker) => Math.max(max, sticker.zIndex),
      0
    );
    
    // 创建新贴纸
    const newSticker: Sticker = {
      id: uuidv4(),
      imageUri: asset.imageUri,
      position,
      scale: 1,
      rotation: 0,
      zIndex: maxZIndex + 1,
      width: asset.width,
      height: asset.height,
      flipped: false
    };
    
    return {
      stickers: [...state.stickers, newSticker],
      selectedStickerId: newSticker.id
    };
  }),
  
  removeSticker: (id) => set(state => ({
    stickers: state.stickers.filter(sticker => sticker.id !== id),
    selectedStickerId: state.selectedStickerId === id ? null : state.selectedStickerId
  })),
  
  clearStickers: () => set({ stickers: [], selectedStickerId: null }),
  
  // 选择操作
  setSelectedStickerId: (id) => set({ selectedStickerId: id }),
  
  // 变换操作
  updateStickerPosition: (id, position) => set(state => ({
    stickers: state.stickers.map(sticker => 
      sticker.id === id 
        ? { ...sticker, position } 
        : sticker
    )
  })),
  
  updateStickerScale: (id, scale) => set(state => ({
    stickers: state.stickers.map(sticker => 
      sticker.id === id 
        ? { ...sticker, scale } 
        : sticker
    )
  })),
  
  updateStickerRotation: (id, rotation) => set(state => ({
    stickers: state.stickers.map(sticker => 
      sticker.id === id 
        ? { ...sticker, rotation } 
        : sticker
    )
  })),
  
  flipSticker: (id) => set(state => ({
    stickers: state.stickers.map(sticker => 
      sticker.id === id 
        ? { ...sticker, flipped: !sticker.flipped } 
        : sticker
    )
  })),
  
  // 层级操作
  bringToFront: (id) => set(state => {
    const maxZIndex = state.stickers.reduce(
      (max, sticker) => Math.max(max, sticker.zIndex),
      0
    );
    
    return {
      stickers: state.stickers.map(sticker => 
        sticker.id === id 
          ? { ...sticker, zIndex: maxZIndex + 1 } 
          : sticker
      )
    };
  }),
  
  sendToBack: (id) => set(state => {
    const minZIndex = state.stickers.reduce(
      (min, sticker) => Math.min(min, sticker.zIndex),
      Number.MAX_SAFE_INTEGER
    );
    
    return {
      stickers: state.stickers.map(sticker => 
        sticker.id === id 
          ? { ...sticker, zIndex: minZIndex - 1 } 
          : sticker
      )
    };
  }),
  
  // 当前分类
  currentCategoryType: StickerCategoryType.EMOJI,
  setCurrentCategoryType: (type) => set({ currentCategoryType: type }),
}));
```

## 贴纸画布组件

贴纸画布组件负责在图像上渲染贴纸，并处理贴纸的点击选择。

```typescript
// src/components/sticker/StickerCanvas.tsx
import React from 'react';
import { View, StyleSheet, Image, TouchableWithoutFeedback } from 'react-native';
import { useStickerStore, Sticker } from '../../stores/stickerStore';
import StickerTransformer from './StickerTransformer';

interface StickerCanvasProps {
  width: number;
  height: number;
}

export default function StickerCanvas({ width, height }: StickerCanvasProps) {
  const { 
    stickers, 
    selectedStickerId, 
    setSelectedStickerId
  } = useStickerStore();
  
  // 按z-index排序贴纸
  const sortedStickers = [...stickers].sort((a, b) => a.zIndex - b.zIndex);
  
  // 取消选中贴纸
  const handleCanvasPress = () => {
    setSelectedStickerId(null);
  };

  return (
    <TouchableWithoutFeedback onPress={handleCanvasPress}>
      <View style={[styles.container, { width, height }]}>
        {/* 渲染所有贴纸 */}
        {sortedStickers.map((sticker) => (
          <React.Fragment key={sticker.id}>
            {/* 普通贴纸 */}
            {selectedStickerId !== sticker.id && (
              <TouchableWithoutFeedback
                onPress={(e) => {
                  e.stopPropagation();
                  setSelectedStickerId(sticker.id);
                }}
              >
                <View
                  style={[
                    styles.stickerContainer,
                    {
                      left: sticker.position.x - (sticker.width * sticker.scale) / 2,
                      top: sticker.position.y - (sticker.height * sticker.scale) / 2,
                      zIndex: sticker.zIndex,
                      transform: [
                        { scale: sticker.scale },
                        { rotate: `${sticker.rotation}deg` },
                        { scaleX: sticker.flipped ? -1 : 1 }
                      ],
                    },
                  ]}
                >
                  <Image
                    source={{ uri: sticker.imageUri }}
                    style={{
                      width: sticker.width,
                      height: sticker.height,
                    }}
                    resizeMode="contain"
                  />
                </View>
              </TouchableWithoutFeedback>
            )}
            
            {/* 选中的贴纸 */}
            {selectedStickerId === sticker.id && (
              <StickerTransformer sticker={sticker} />
            )}
          </React.Fragment>
        ))}
      </View>
    </TouchableWithoutFeedback>
  );
}

const styles = StyleSheet.create({
  container: {
    position: 'absolute',
    top: 0,
    left: 0,
  },
  stickerContainer: {
    position: 'absolute',
    justifyContent: 'center',
    alignItems: 'center',
  },
});

## 贴纸变换控制器

贴纸变换控制器为选中的贴纸提供调整大小、旋转和位置的功能。

```typescript
// src/components/sticker/StickerTransformer.tsx
import React, { useRef, useState } from 'react';
import {
  View,
  StyleSheet,
  Image,
  PanResponder,
  GestureResponderEvent,
  PanResponderGestureState,
  Animated,
} from 'react-native';
import { useStickerStore, Sticker } from '../../stores/stickerStore';
import { Ionicons } from '@expo/vector-icons';

interface StickerTransformerProps {
  sticker: Sticker;
}

export default function StickerTransformer({ sticker }: StickerTransformerProps) {
  const {
    updateStickerPosition,
    updateStickerScale,
    updateStickerRotation,
    removeSticker,
    bringToFront,
    flipSticker,
  } = useStickerStore();
  
  // 动画值
  const pan = useRef(new Animated.ValueXY({ x: 0, y: 0 })).current;
  
  // 初始位置和旋转
  const [initialPosition, setInitialPosition] = useState({ x: 0, y: 0 });
  const [initialDistance, setInitialDistance] = useState(0);
  const [initialScale, setInitialScale] = useState(1);
  const [initialAngle, setInitialAngle] = useState(0);
  const [initialRotation, setInitialRotation] = useState(0);
  
  // 计算两个触摸点之间的距离
  const distance = (x1: number, y1: number, x2: number, y2: number) => {
    return Math.sqrt(Math.pow(x2 - x1, 2) + Math.pow(y2 - y1, 2));
  };
  
  // 计算两个触摸点之间的角度
  const angle = (x1: number, y1: number, x2: number, y2: number) => {
    return Math.atan2(y2 - y1, x2 - x1) * 180 / Math.PI;
  };
  
  // 处理移动手势
  const panResponder = useRef(
    PanResponder.create({
      onStartShouldSetPanResponder: () => true,
      onMoveShouldSetPanResponder: () => true,
      onPanResponderGrant: (e) => {
        // 将贴纸带到前面
        bringToFront(sticker.id);
        
        // 存储初始位置
        setInitialPosition({
          x: sticker.position.x,
          y: sticker.position.y,
        });
        
        // 如果是两个触摸点，则存储初始缩放和旋转
        if (e.nativeEvent.touches.length === 2) {
          const touch1 = e.nativeEvent.touches[0];
          const touch2 = e.nativeEvent.touches[1];
          
          setInitialDistance(
            distance(touch1.pageX, touch1.pageY, touch2.pageX, touch2.pageY)
          );
          
          setInitialAngle(
            angle(touch1.pageX, touch1.pageY, touch2.pageX, touch2.pageY)
          );
          
          setInitialScale(sticker.scale);
          setInitialRotation(sticker.rotation);
        }
      },
      onPanResponderMove: (e: GestureResponderEvent, gestureState: PanResponderGestureState) => {
        // 如果是单指拖动，则移动贴纸
        if (e.nativeEvent.touches.length === 1) {
          updateStickerPosition(sticker.id, {
            x: initialPosition.x + gestureState.dx,
            y: initialPosition.y + gestureState.dy,
          });
        }
        // 如果是双指操作，则缩放和旋转贴纸
        else if (e.nativeEvent.touches.length === 2) {
          const touch1 = e.nativeEvent.touches[0];
          const touch2 = e.nativeEvent.touches[1];
          
          const currentDistance = distance(
            touch1.pageX, touch1.pageY, touch2.pageX, touch2.pageY
          );
          
          const currentAngle = angle(
            touch1.pageX, touch1.pageY, touch2.pageX, touch2.pageY
          );
          
          // 更新缩放
          const newScale = initialScale * (currentDistance / initialDistance);
          if (newScale > 0.3 && newScale < 3) {
            updateStickerScale(sticker.id, newScale);
          }
          
          // 更新旋转
          const angleDiff = currentAngle - initialAngle;
          updateStickerRotation(sticker.id, initialRotation + angleDiff);
        }
      },
      onPanResponderRelease: () => {
        // 重置动画值
        pan.flattenOffset();
      },
    })
  ).current;
  
  // 处理删除贴纸
  const handleDelete = () => {
    removeSticker(sticker.id);
  };
  
  // 处理翻转贴纸
  const handleFlip = () => {
    flipSticker(sticker.id);
  };

  return (
    <View
      style={[
        styles.container,
        {
          left: sticker.position.x - (sticker.width * sticker.scale) / 2,
          top: sticker.position.y - (sticker.height * sticker.scale) / 2,
          transform: [
            { scale: sticker.scale },
            { rotate: `${sticker.rotation}deg` },
            { scaleX: sticker.flipped ? -1 : 1 }
          ],
        },
      ]}
      {...panResponder.panHandlers}
    >
      {/* 贴纸图像 */}
      <Image
        source={{ uri: sticker.imageUri }}
        style={{
          width: sticker.width,
          height: sticker.height,
        }}
        resizeMode="contain"
      />
      
      {/* 控制按钮 */}
      <View style={styles.deleteButton} onTouchEnd={handleDelete}>
        <Ionicons name="close-circle" size={24} color="#e74c3c" />
      </View>
      
      <View style={styles.flipButton} onTouchEnd={handleFlip}>
        <Ionicons name="swap-horizontal" size={22} color="#3498db" />
      </View>
      
      <View style={styles.rotateButton}>
        <Ionicons name="refresh" size={22} color="#3498db" />
      </View>
      
      <View style={styles.scaleButton}>
        <Ionicons name="resize" size={22} color="#3498db" />
      </View>
      
      {/* 边框 */}
      <View style={styles.border} />
    </View>
  );
}

## 贴纸模块集成与总结

贴纸模块通过以下方式集成到编辑器中：

1. 当用户切换到贴纸编辑类别时，显示贴纸面板：

```typescript
// 在EditorPanels.tsx中
switch (activeCategoryId) {
  // ... 其他类别
  case MainCategoryType.STICKER:
    return <StickerPanel />;
  // ... 其他类别
}
```

2. 不论用户处于哪个编辑模式，贴纸都会显示在画布上，并且可以进行交互。

3. 用户可以通过点击贴纸面板中的贴纸来添加它们到画布上，然后通过StickerTransformer进行移动、缩放、旋转和翻转操作。

贴纸模块具有以下特性：

1. **分类管理**：贴纸按类别进行组织和展示，包括表情、装饰、文字、形状和自定义类别
2. **贴纸添加**：允许用户从贴纸库中选择贴纸并添加到图像上
3. **贴纸变换**：
   - 移动：通过拖动贴纸可以将其移动到任意位置
   - 缩放：可以通过双指缩放或使用缩放控制点来调整贴纸大小
   - 旋转：可以通过双指旋转或使用旋转控制点来旋转贴纸
   - 翻转：可以水平翻转贴纸，创造镜像效果
4. **层级管理**：
   - 前移：将贴纸移到最前面，显示在其他贴纸上方
   - 后移：将贴纸移到最后面，显示在其他贴纸下方
   - 删除：删除选中的贴纸
   - 清除：删除所有贴纸

这种实现让用户可以创建丰富的图像组合，添加个性化元素，使照片更有趣和有表现力。

```typescript
// src/components/editor/EditorCanvas.tsx
import React, { useState, useEffect } from 'react';
import { View, StyleSheet, Dimensions } from 'react-native';
import { Image } from 'expo-image';
import { Canvas, useImage } from '@shopify/react-native-skia';
import { useEditorStore } from '../../stores/editorStore';
import { useFilterStore } from '../../stores/filterStore';
import { MainCategoryType } from '../../types/editorTypes';
import ImageFilters from '../filter/ImageFilters';
import DoodleCanvas from '../doodle/DoodleCanvas';
import StickerCanvas from '../sticker/StickerCanvas';

interface EditorCanvasProps {
  imageUri: string;
}

export default function EditorCanvas({ imageUri }: EditorCanvasProps) {
  const [canvasDimensions, setCanvasDimensions] = useState({ width: 0, height: 0 });
  const screenWidth = Dimensions.get('window').width;
  const screenHeight = Dimensions.get('window').height;
  
  const { activeCategoryId, setImageSize } = useEditorStore();
  const { 
    selectedLutId, 
    lutIntensity,
    brightness,
    contrast,
    saturation
  } = useFilterStore();
  
  // 加载图像和LUT
  const image = useImage(imageUri);
  const lutImage = selectedLutId ? useImage(`assets/luts/${selectedLutId}.png`) : null;
  
  // 计算画布尺寸
  useEffect(() => {
    if (imageUri) {
      Image.getSize(imageUri, (width, height) => {
        const aspectRatio = width / height;
        const maxWidth = screenWidth * 0.9;
        const maxHeight = screenHeight * 0.5;
        
        let canvasWidth, canvasHeight;
        
        if (width > height) {
          // 横向图片
          canvasWidth = Math.min(maxWidth, width);
          canvasHeight = canvasWidth / aspectRatio;
          
          if (canvasHeight > maxHeight) {
            canvasHeight = maxHeight;
            canvasWidth = canvasHeight * aspectRatio;
          }
        } else {
          // 纵向图片
          canvasHeight = Math.min(maxHeight, height);
          canvasWidth = canvasHeight * aspectRatio;
          
          if (canvasWidth > maxWidth) {
            canvasWidth = maxWidth;
            canvasHeight = canvasWidth / aspectRatio;
          }
        }
        
        setCanvasDimensions({
          width: canvasWidth,
          height: canvasHeight
        });
        
        // 更新全局状态中的图像尺寸
        setImageSize({ width: canvasWidth, height: canvasHeight });
      });
    }
  }, [imageUri, screenWidth, screenHeight, setImageSize]);
  
  // 根据当前激活的编辑类别渲染不同的画布内容
  const renderCanvasContent = () => {
    if (!canvasDimensions.width || !canvasDimensions.height) {
      return null;
    }
    
    // 创建基础画布容器
    const canvasContainer = (
      <View style={[styles.canvas, { width: canvasDimensions.width, height: canvasDimensions.height }]}>
        {!image ? (
          <Image
            source={{ uri: imageUri }}
            style={{ width: '100%', height: '100%' }}
            contentFit="contain"
          />
        ) : (
          <Canvas style={{ width: canvasDimensions.width, height: canvasDimensions.height }}>
            <ImageFilters
              image={image}
              lutImage={lutImage}
              intensity={lutIntensity}
              brightness={brightness}
              contrast={contrast}
              saturation={saturation}
              width={canvasDimensions.width}
              height={canvasDimensions.height}
            />
          </Canvas>
        )}
        
        {/* 贴纸画布 - 始终显示 */}
        <StickerCanvas 
          width={canvasDimensions.width} 
          height={canvasDimensions.height} 
        />
      </View>
    );
    
    // 涂鸦模式 - 在贴纸上方渲染涂鸦画布
    if (activeCategoryId === MainCategoryType.DOODLE) {
      return (
        <View style={[styles.canvas, { width: canvasDimensions.width, height: canvasDimensions.height }]}>
          {canvasContainer}
          <DoodleCanvas 
            width={canvasDimensions.width} 
            height={canvasDimensions.height} 
          />
        </View>
      );
    }
    
    // 其他编辑模式
    return canvasContainer;
  };
  
  return (
    <View style={styles.container}>
      {renderCanvasContent()}
    </View>
  );
}

    overflow: 'hidden',
    borderRadius: 8,
    // 添加阴影效果
    shadowColor: '#000',
    shadowOffset: { width: 0, height: 2 },
    shadowOpacity: 0.1,
    shadowRadius: 4,
    elevation: 3,
  },
});
