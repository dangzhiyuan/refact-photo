# 涂鸦模块实现

涂鸦模块提供手指绘画功能，允许用户在图像上自由绘制线条、形状和图案。该功能模块包含以下核心功能：

1. 自由绘画工具
2. 多种笔刷类型和粗细
3. 颜色选择系统
4. 撤销/重做操作
5. 图层管理

## 模块结构

涂鸦模块由以下主要组件构成：

```
doodle/
├── DoodlePanel.tsx               // 涂鸦面板主容器
├── DoodleCanvas.tsx              // 涂鸦绘制画布
├── DoodleToolbar.tsx             // 绘画工具栏
├── DoodleBrushSelector.tsx       // 笔刷类型选择器
├── DoodleColorPicker.tsx         // 颜色选择器
├── DoodleStrokeWidthSelector.tsx // 线条粗细选择器
└── DoodleHistoryControls.tsx     // 历史记录控制(撤销/重做)
```

## 涂鸦状态管理

涂鸦功能需要专门的状态管理，用于跟踪绘画路径、选定的工具和颜色等。

```typescript
// src/stores/doodleStore.ts
import { create } from 'zustand';
import { v4 as uuidv4 } from 'uuid';
import { DoodleBrushType, DoodleColor, Point } from '../types/editorTypes';

interface DoodlePath {
  id: string;
  points: Point[];
  color: DoodleColor;
  brushType: DoodleBrushType;
  strokeWidth: number;
}

interface DoodleState {
  // 路径管理
  paths: DoodlePath[];
  currentPath: Point[] | null;
  addPath: (path: Point[], color: DoodleColor, brushType: DoodleBrushType, strokeWidth: number) => void;
  updateCurrentPath: (point: Point) => void;
  startNewPath: (point: Point) => void;
  endCurrentPath: () => void;
  clearPaths: () => void;
  
  // 工具设置
  selectedColor: DoodleColor;
  setSelectedColor: (color: DoodleColor) => void;
  
  selectedBrushType: DoodleBrushType;
  setSelectedBrushType: (type: DoodleBrushType) => void;
  
  strokeWidth: number;
  setStrokeWidth: (width: number) => void;
  
  // 历史记录
  undoPath: () => void;
  redoPath: () => void;
  canUndo: boolean;
  canRedo: boolean;
  
  // 已删除的路径（用于重做）
  deletedPaths: DoodlePath[];
}

export const useDoodleStore = create<DoodleState>((set, get) => ({
  // 路径管理
  paths: [],
  currentPath: null,
  addPath: (points, color, brushType, strokeWidth) => set(state => ({
    paths: [...state.paths, {
      id: uuidv4(),
      points,
      color,
      brushType,
      strokeWidth
    }],
    deletedPaths: [],
    canUndo: true,
    canRedo: false,
  })),
  updateCurrentPath: (point) => set(state => {
    if (!state.currentPath) return state;
    return { currentPath: [...state.currentPath, point] };
  }),
  startNewPath: (point) => set({ currentPath: [point] }),
  endCurrentPath: () => {
    const { currentPath, selectedColor, selectedBrushType, strokeWidth } = get();
    if (currentPath && currentPath.length > 1) {
      get().addPath(currentPath, selectedColor, selectedBrushType, strokeWidth);
    }
    set({ currentPath: null });
  },
  clearPaths: () => set({ 
    paths: [], 
    deletedPaths: [],
    canUndo: false,
    canRedo: false 
  }),
  
  // 工具设置
  selectedColor: { r: 0, g: 0, b: 0, a: 1 }, // 默认黑色
  setSelectedColor: (color) => set({ selectedColor: color }),
  
  selectedBrushType: DoodleBrushType.NORMAL,
  setSelectedBrushType: (type) => set({ selectedBrushType: type }),
  
  strokeWidth: 5,
  setStrokeWidth: (width) => set({ strokeWidth: width }),
  
  // 历史记录
  canUndo: false,
  canRedo: false,
  undoPath: () => set(state => {
    if (state.paths.length === 0) return state;
    
    const lastPath = state.paths[state.paths.length - 1];
    const newPaths = state.paths.slice(0, -1);
    
    return {
      paths: newPaths,
      deletedPaths: [...state.deletedPaths, lastPath],
      canUndo: newPaths.length > 0,
      canRedo: true
    };
  }),
  redoPath: () => set(state => {
    if (state.deletedPaths.length === 0) return state;
    
    const pathToRestore = state.deletedPaths[state.deletedPaths.length - 1];
    const newDeletedPaths = state.deletedPaths.slice(0, -1);
    
    return {
      paths: [...state.paths, pathToRestore],
      deletedPaths: newDeletedPaths,
      canUndo: true,
      canRedo: newDeletedPaths.length > 0
    };
  }),
  
  deletedPaths: [],
}));
```

## 涂鸦面板实现

涂鸦面板是涂鸦模块的主要入口，整合了所有涂鸦相关组件。

```typescript
// src/components/doodle/DoodlePanel.tsx
import React from 'react';
import { View, StyleSheet } from 'react-native';
import { useEditorStore } from '../../stores/editorStore';
import { useDoodleStore } from '../../stores/doodleStore';
import DoodleToolbar from './DoodleToolbar';
import DoodleBrushSelector from './DoodleBrushSelector';
import DoodleColorPicker from './DoodleColorPicker';
import DoodleStrokeWidthSelector from './DoodleStrokeWidthSelector';
import DoodleHistoryControls from './DoodleHistoryControls';

export default function DoodlePanel() {
  const { currentImageUri } = useEditorStore();
  const { 
    selectedBrushType, 
    setSelectedBrushType,
    selectedColor,
    setSelectedColor,
    strokeWidth,
    setStrokeWidth,
    undoPath,
    redoPath,
    clearPaths,
    canUndo,
    canRedo
  } = useDoodleStore();

  if (!currentImageUri) {
    return null;
  }

  return (
    <View style={styles.container}>
      {/* 顶部工具栏 */}
      <DoodleToolbar
        onUndo={undoPath}
        onRedo={redoPath}
        onClear={clearPaths}
        canUndo={canUndo}
        canRedo={canRedo}
      />
      
      {/* 笔刷选择器 */}
      <DoodleBrushSelector
        selectedBrushType={selectedBrushType}
        onChange={setSelectedBrushType}
      />
      
      {/* 底部控制面板 */}
      <View style={styles.controlsContainer}>
        <DoodleColorPicker
          selectedColor={selectedColor}
          onChange={setSelectedColor}
        />
        
        <DoodleStrokeWidthSelector
          value={strokeWidth}
          onChange={setStrokeWidth}
        />
      </View>
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    width: '100%',
    paddingBottom: 16,
  },
  controlsContainer: {
    marginTop: 16,
    paddingHorizontal: 16,
  },
});
```

```typescript
  selectedBrushType,
  onChange
}: DoodleBrushSelectorProps) {
  // 笔刷选项
  const brushOptions: BrushOption[] = [
    { type: DoodleBrushType.NORMAL, icon: 'brush', label: '普通' },
    { type: DoodleBrushType.BLACK_BORDER, icon: 'brush-outline', label: '描边' },

    { type: DoodleBrushType.SPARKLER, icon: 'sparkles', label: '闪光' },
  ];

  return (
    <View style={styles.container}>
      <ScrollView
        horizontal
        showsHorizontalScrollIndicator={false}
        contentContainerStyle={styles.scrollContent}
      >
        {brushOptions.map((brush) => (
          <TouchableOpacity
            key={brush.type}
            style={[
              styles.brushItem,
              selectedBrushType === brush.type && styles.selectedBrushItem

              size={24}

              ]}
            >
              {brush.label}
            </Text>
          </TouchableOpacity>
        ))}
```

```typescript
    onChange(color);
```

```typescript
            size={22}
            color={canRedo ? '#3498db' : '#bdc3c7'}
```

## 涂鸦画布组件

核心绘画组件，使用React Native Skia实现高性能绘图功能。

```typescript
// src/components/doodle/DoodleCanvas.tsx
import React from 'react';
import { StyleSheet } from 'react-native';
import { 
  Canvas, 
  Path, 
  useTouchHandler, 
  SkPath, 
  Skia,
  useValue,
  useComputedValue,
  PathProps
} from '@shopify/react-native-skia';
import { useDoodleStore } from '../../stores/doodleStore';
import { DoodleBrushType, DoodleColor, Point } from '../../types/editorTypes';

interface DoodleCanvasProps {
  width: number;
  height: number;
}

export default function DoodleCanvas({ width, height }: DoodleCanvasProps) {
  const {
    paths,
    currentPath,
    startNewPath,
    updateCurrentPath,
    endCurrentPath,
    selectedColor,
    selectedBrushType,
    strokeWidth
  } = useDoodleStore();
  
  // 当前路径的Skia Path对象
  const currentSkPath = useValue<SkPath | null>(null);
  
  // 创建一个新的路径
  const createNewPath = (point: Point) => {
    const path = Skia.Path.Make();
    path.moveTo(point.x, point.y);
    currentSkPath.current = path;
    startNewPath(point);
  };
  
  // 触摸处理函数
  const touchHandler = useTouchHandler({
    onStart: ({ x, y }) => {
      createNewPath({ x, y });
    },
    onActive: ({ x, y }) => {
      if (currentSkPath.current) {
        const newPoint = { x, y };
        currentSkPath.current.lineTo(x, y);
        updateCurrentPath(newPoint);
      }
    },
    onEnd: () => {
      endCurrentPath();
      currentSkPath.current = null;
    },
    onCancel: () => {
      endCurrentPath();
      currentSkPath.current = null;
    }
  });
  
  // 计算当前路径的SVG路径字符串
  const currentPathSvg = useComputedValue(() => {
    if (!currentPath || currentPath.length < 2 || !currentSkPath.current) {
      return '';
    }
    return currentSkPath.current.toSVGString();
  }, [currentPath]);
  
  // 获取笔刷样式
  const getBrushStyle = (
    color: DoodleColor, 
    brushType: DoodleBrushType,
    width: number
  ): Partial<PathProps> => {
    const rgbaColor = `rgba(${color.r}, ${color.g}, ${color.b}, ${color.a})`;
    
    switch (brushType) {
      case DoodleBrushType.BLACK_BORDER:
        return {
          color: 'black',
          style: 'stroke',
          strokeWidth: width + 4,
        };
      case DoodleBrushType.WHITE_BODY:
        return {
          color: 'white',
          style: 'stroke',
          strokeWidth: width,
        };
      case DoodleBrushType.TRANSLUCENT:
        return {
          color: `rgba(${color.r}, ${color.g}, ${color.b}, 0.3)`,
          style: 'stroke',
          strokeWidth: width,
        };
      case DoodleBrushType.BLUR_BORDER_WHITE_BODY:
        return {
          color: rgbaColor,
          style: 'stroke',
          strokeWidth: width,
          strokeJoin: 'round',
          strokeCap: 'round',
          // 在完整实现中还可以添加模糊效果等
        };
      case DoodleBrushType.COLOR_BAR:
        return {
          color: rgbaColor,
          style: 'stroke',
          strokeWidth: width * 2,
          strokeJoin: 'miter',
          strokeCap: 'square',
        };
      case DoodleBrushType.SPARKLER:
        return {
          color: rgbaColor,
          style: 'stroke',
          strokeWidth: width * 0.8,
          // 在完整实现中可以添加渐变和闪光效果
        };
      case DoodleBrushType.NORMAL:
      default:
        return {
          color: rgbaColor,
          style: 'stroke',
          strokeWidth: width,
          strokeJoin: 'round',
          strokeCap: 'round',
        };
    }
  };
  
  // 将点数组转换为SVG路径字符串
  const pointsToPath = (points: Point[]): string => {
    if (points.length < 2) return '';
    
    const [first, ...rest] = points;
    let path = `M ${first.x} ${first.y}`;
    
    rest.forEach(point => {
      path += ` L ${point.x} ${point.y}`;
    });
    
    return path;
  };

  return (
    <Canvas style={[styles.canvas, { width, height }]} onTouch={touchHandler}>
      {/* 已保存的路径 */}
      {paths.map(path => {
        const { id, points, color, brushType, strokeWidth: pathWidth } = path;
        const pathString = pointsToPath(points);
        const brushStyle = getBrushStyle(color, brushType, pathWidth);
        
        // 对于某些特殊笔刷类型，需要渲染多层
        if (brushType === DoodleBrushType.BLACK_BORDER) {
          const baseStyle = getBrushStyle(color, DoodleBrushType.NORMAL, pathWidth);
          return (
            <React.Fragment key={id}>
              <Path
                path={pathString}
                color={brushStyle.color}
                style="stroke"
                strokeWidth={brushStyle.strokeWidth}
                strokeJoin="round"
                strokeCap="round"
              />
              <Path
                path={pathString}
                color={baseStyle.color}
                style="stroke"
                strokeWidth={baseStyle.strokeWidth as number}
                strokeJoin="round"
                strokeCap="round"
              />
            </React.Fragment>
          );
        }
        
        return (
          <Path
            key={id}
            path={pathString}
            color={brushStyle.color}
            style={brushStyle.style}
            strokeWidth={brushStyle.strokeWidth}
            strokeJoin={brushStyle.strokeJoin || 'round'}
            strokeCap={brushStyle.strokeCap || 'round'}
          />
        );
      })}
      
      {/* 当前绘制中的路径 */}
      {currentPath && currentPath.length >= 2 && (
        <Path
          path={currentPathSvg}
          {...getBrushStyle(selectedColor, selectedBrushType, strokeWidth)}
        />
      )}
    </Canvas>
  );
}

const styles = StyleSheet.create({
  canvas: {
    backgroundColor: 'transparent',
  },
});
```

```typescript
  const screenWidth = Dimensions.get('window').width;
  const screenHeight = Dimensions.get('window').height;
  
  const { activeCategoryId, setImageSize } = useEditorStore();
  const { 
    selectedLutId, 
    lutIntensity,
    brightness,
    contrast,
    saturation
  } = useFilterStore();

    if (imageUri) {
      Image.getSize(imageUri, (width, height) => {
        const aspectRatio = width / height;
        const maxWidth = screenWidth * 0.9;
        const maxHeight = screenHeight * 0.5;
        
        let canvasWidth, canvasHeight;
        
        if (width > height) {
          // 横向图片
          canvasWidth = Math.min(maxWidth, width);
          canvasHeight = canvasWidth / aspectRatio;
          
          if (canvasHeight > maxHeight) {
            canvasHeight = maxHeight;
            canvasWidth = canvasHeight * aspectRatio;
          }
        } else {
          // 纵向图片
          canvasHeight = Math.min(maxHeight, height);
          canvasWidth = canvasHeight * aspectRatio;
          
          if (canvasWidth > maxWidth) {
            canvasWidth = maxWidth;
            canvasHeight = canvasWidth / aspectRatio;
          }
        }
        
        setCanvasDimensions({
          width: canvasWidth,
          height: canvasHeight
        });
        
        // 更新全局状态中的图像尺寸
        setImageSize({ width: canvasWidth, height: canvasHeight });
      });
    }
  }, [imageUri, screenWidth, screenHeight, setImageSize]);

              intensity={lutIntensity}
              brightness={brightness}
              contrast={contrast}
              saturation={saturation}
              width={canvasDimensions.width}
              height={canvasDimensions.height}
            />
          </Canvas>
        )}
      </View>
    );
    
    // 涂鸦模式
    if (activeCategoryId === MainCategoryType.DOODLE) {
      return (
        <View style={[styles.canvas, { width: canvasDimensions.width, height: canvasDimensions.height }]}>
          {canvasContainer}
          <DoodleCanvas 
            width={canvasDimensions.width} 
            height={canvasDimensions.height} 
          />
        </View>
      );
    }
    
    // 其他编辑模式
    return canvasContainer;
  };
  
  return (
    <View style={styles.container}>
      {renderCanvasContent()}
    </View>
  );
}
```

## 路径优化工具

用于优化绘制路径，提高性能和呈现效果。

```typescript
// src/utils/PathOptimizer.ts
import { Point } from '../types/editorTypes';

/**
 * 路径优化工具类
 * 用于简化和平滑手指绘制的路径，提高渲染性能
 */
class PathOptimizer {
  /**
   * 简化路径点，减少点的数量但保持路径形状
   * 使用道格拉斯-普克算法(Douglas-Peucker algorithm)
   */
  simplifyPath(points: Point[], tolerance: number = 2): Point[] {
    if (points.length <= 2) return points;
    
    // 查找最远点的索引和距离
    const findFurthestPointIndex = (start: Point, end: Point, points: Point[]): [number, number] => {
      let maxDistance = 0;
      let index = 0;
      
      for (let i = 1; i < points.length - 1; i++) {
        const distance = this.perpendicularDistance(points[i], start, end);
        if (distance > maxDistance) {
          maxDistance = distance;
          index = i;
        }
      }
      
      return [index, maxDistance];
    };
    
    // 递归简化
    const douglasPeucker = (points: Point[], startIndex: number, endIndex: number): Point[] => {
      if (endIndex - startIndex <= 1) {
        return [points[startIndex]];
      }
      
      const [furthestIndex, maxDistance] = findFurthestPointIndex(
        points[startIndex], 
        points[endIndex], 
        points.slice(startIndex, endIndex + 1)
      );
      
      const actualIndex = startIndex + furthestIndex;
      
      if (maxDistance > tolerance) {
        // 递归处理两个子路径
        const leftPath = douglasPeucker(points, startIndex, actualIndex);
        const rightPath = douglasPeucker(points, actualIndex, endIndex);
        
        // 合并结果但避免重复点
        return [...leftPath, ...rightPath.slice(1)];
      } else {
        // 距离小于容差，直接返回端点
        return [points[startIndex], points[endIndex]];
      }
    };
    
    return douglasPeucker(points, 0, points.length - 1);
  }
  
  /**
   * 计算点到线段的垂直距离
   */
  private perpendicularDistance(point: Point, lineStart: Point, lineEnd: Point): number {
    const dx = lineEnd.x - lineStart.x;
    const dy = lineEnd.y - lineStart.y;
    
    // 线段长度平方
    const lineLengthSquared = dx * dx + dy * dy;
    
    if (lineLengthSquared === 0) {
      // 线段是一个点
      return this.distanceBetweenPoints(point, lineStart);
    }
    
    // 计算投影参数 t
    const t = ((point.x - lineStart.x) * dx + (point.y - lineStart.y) * dy) / lineLengthSquared;
    
    if (t < 0) {
      // 投影点在线段起点之前
      return this.distanceBetweenPoints(point, lineStart);
    }
    
    if (t > 1) {
      // 投影点在线段终点之后
      return this.distanceBetweenPoints(point, lineEnd);
    }
    
    // 投影点在线段上
    const projectionX = lineStart.x + t * dx;
    const projectionY = lineStart.y + t * dy;
    
    return this.distanceBetweenPoints(point, { x: projectionX, y: projectionY });
  }
  
  /**
   * 计算两点之间的距离
   */
  private distanceBetweenPoints(p1: Point, p2: Point): number {
    const dx = p2.x - p1.x;
    const dy = p2.y - p1.y;
    return Math.sqrt(dx * dx + dy * dy);
  }
  
  /**
   * 平滑路径，使用贝塞尔曲线
   * 这个实现是简化版，实际应用中可能需要更复杂的算法
   */
  smoothPath(points: Point[], tension: number = 0.5): string {
    if (points.length < 3) {
      return this.pointsToSvgPath(points);
    }
    
    let path = `M ${points[0].x} ${points[0].y}`;
    
    for (let i = 0; i < points.length - 2; i++) {
      const p0 = i > 0 ? points[i - 1] : points[0];
      const p1 = points[i];
      const p2 = points[i + 1];
      const p3 = i < points.length - 2 ? points[i + 2] : p2;
      
      // 计算控制点
      const cp1x = p1.x + (p2.x - p0.x) * tension;
      const cp1y = p1.y + (p2.y - p0.y) * tension;
      const cp2x = p2.x - (p3.x - p1.x) * tension;
      const cp2y = p2.y - (p3.y - p1.y) * tension;
      
      // 添加贝塞尔曲线
      path += ` C ${cp1x} ${cp1y}, ${cp2x} ${cp2y}, ${p2.x} ${p2.y}`;
    }
    
    return path;
  }
  
  /**
   * 将点数组转换为SVG路径
   */
  pointsToSvgPath(points: Point[]): string {
    if (points.length === 0) return '';
    
    let path = `M ${points[0].x} ${points[0].y}`;
    
    for (let i = 1; i < points.length; i++) {
      path += ` L ${points[i].x} ${points[i].y}`;
    }
    
    return path;
  }
}

export default new PathOptimizer();
```
