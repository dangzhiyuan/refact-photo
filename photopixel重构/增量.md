```typescript
// src/components/editor/EditorCanvas.tsx
import React, { useState, useEffect } from 'react';
import { View, StyleSheet, Dimensions } from 'react-native';
import { Image } from 'expo-image';
import { GestureDetector, Gesture } from 'react-native-gesture-handler';
import Animated, { useSharedValue, useAnimatedStyle } from 'react-native-reanimated';
import { useEditorStore } from '../../stores/editorStore';
import FilterCanvasFill from '../filter/FilterCanvasFill';

interface EditorCanvasProps {
  imageUri: string;
  parentScreenWidth: number;
  parentScreenHeight: number;
}

export default function EditorCanvas({ 
  imageUri, 
  parentScreenWidth, 
  parentScreenHeight 
}: EditorCanvasProps) {
  const [canvasDimensions, setCanvasDimensions] = useState({ width: 0, height: 0 });
  const { selectedLut, activeCategoryId } = useEditorStore();
  
  // 计算画布尺寸
  useEffect(() => {
    if (imageUri) {
      Image.getSize(imageUri, (width, height) => {
        const aspectRatio = width / height;
        const maxWidth = parentScreenWidth * 0.9;
        const maxHeight = parentScreenHeight * 0.6;
        
        let canvasWidth, canvasHeight;
        
        if (width > height) {
          canvasWidth = Math.min(maxWidth, width);
          canvasHeight = canvasWidth / aspectRatio;
          
          if (canvasHeight > maxHeight) {
            canvasHeight = maxHeight;
            canvasWidth = canvasHeight * aspectRatio;
          }
        } else {
          canvasHeight = Math.min(maxHeight, height);
          canvasWidth = canvasHeight * aspectRatio;
          
          if (canvasWidth > maxWidth) {
            canvasWidth = maxWidth;
            canvasHeight = canvasWidth / aspectRatio;
          }
        }
        
        setCanvasDimensions({
          width: canvasWidth,
          height: canvasHeight
        });
      });
    }
  }, [imageUri, parentScreenWidth, parentScreenHeight]);
  
  if (!canvasDimensions.width || !canvasDimensions.height) {
    return <View style={styles.placeholder} />;
  }
  
  return (
    <View style={styles.container}>
      <FilterCanvasFill
        imageUri={imageUri}
        width={canvasDimensions.width}
        height={canvasDimensions.height}
      />
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    alignItems: 'center',
    justifyContent: 'center',
  },
  placeholder: {
    width: 200,
    height: 200,
    backgroundColor: '#f0f0f0',
  },
});
```

```typescript
// src/utils/NavigationUtil.ts
import { createNavigationContainerRef } from '@react-navigation/native';
import { RootStackParamList } from '../navigation/AppNavigator';

export const navigationRef = createNavigationContainerRef<RootStackParamList>();

export default {
  navigate<RouteName extends keyof RootStackParamList>(
    name: RouteName,
    params?: RootStackParamList[RouteName]
  ) {
    if (navigationRef.isReady()) {
      navigationRef.navigate(name, params);
    }
  },
  
  goBack() {
    if (navigationRef.isReady() && navigationRef.canGoBack()) {
      navigationRef.goBack();
    }
  },
  
  reset(index = 0, routes: any[]) {
    if (navigationRef.isReady()) {
      navigationRef.reset({
        index,
        routes,
      });
    }
  }
};
```

```typescript
// src/stores/stickerStore.ts
import create from 'zustand';
import { v4 as uuidv4 } from 'uuid';
import { StickerType } from '../types/editorTypes';

export interface Sticker {
  id: string;
  type: StickerType;
  resourceKey: string;
}

interface StickerState {
  stickers: Sticker[];
  selectedStickerId: string | null;
  
  addSticker: (type: StickerType, resourceKey: string) => void;
  removeSticker: (id: string) => void;
  clearStickers: () => void;
  setSelectedStickerId: (id: string | null) => void;
}

export const useStickerStore = create<StickerState>((set) => ({
  stickers: [],
  selectedStickerId: null,
  
  addSticker: (type, resourceKey) => set((state) => {
    const newSticker = {
      id: uuidv4(),
      type,
      resourceKey
    };
    return { stickers: [...state.stickers, newSticker] };
  }),
  
  removeSticker: (id) => set((state) => ({
    stickers: state.stickers.filter(sticker => sticker.id !== id),
    selectedStickerId: state.selectedStickerId === id ? null : state.selectedStickerId
  })),
  
  clearStickers: () => set({ stickers: [], selectedStickerId: null }),
  
  setSelectedStickerId: (id) => set({ selectedStickerId: id }),
}));
```

```typescript
import React from 'react';
import { 
  TouchableOpacity,

interface ButtonProps {
  title: string;
  onPress: () => void;
  style?: ViewStyle;
  textStyle?: TextStyle;
  disabled?: boolean;
  loading?: boolean;
  variant?: 'primary' | 'secondary' | 'outline';
}

export default function Button({ 
  title, 
  onPress, 
  style, 
  textStyle, 
  disabled = false,
  loading = false,
  variant = 'primary'
}: ButtonProps) {
  const getButtonStyle = () => {
    switch(variant) {
      case 'primary':
        return [styles.button, styles.primaryButton, style];
      case 'secondary':
        return [styles.button, styles.secondaryButton, style];
      case 'outline':
        return [styles.button, styles.outlineButton, style];
      default:
        return [styles.button, styles.primaryButton, style];
    }
  };

  const getTextStyle = () => {
    switch(variant) {
      case 'primary':
        return [styles.text, styles.primaryText, textStyle];
      case 'secondary':
        return [styles.text, styles.secondaryText, textStyle];
      case 'outline':
        return [styles.text, styles.outlineText, textStyle];
      default:
        return [styles.text, styles.primaryText, textStyle];
    }
  };

  return (
    <TouchableOpacity
      style={[getButtonStyle(), disabled && styles.disabledButton]}
      onPress={onPress}
      disabled={disabled || loading}
    >
      {loading ? (
        <ActivityIndicator color={variant === 'outline' ? '#3498db' : '#fff'} />
      ) : (
        <Text style={[getTextStyle(), disabled && styles.disabledText]}>
```

## 媒体库访问实现

相册访问是应用的核心功能之一，使用expo-media-library实现：

```typescript
// src/hooks/useAssets.ts
import { useEffect, useState } from 'react';
import * as MediaLibrary from 'expo-media-library';

interface UseAssetsProps {
  assets: MediaLibrary.Asset[];
  loading: boolean;
  hasNextPage: boolean;
  loadMore: () => Promise<void>;
  refreshAssets: () => Promise<void>;
  error: Error | null;
}

export function useAssets(
  mediaType: MediaLibrary.MediaTypeValue = MediaLibrary.MediaType.photo,
  limit: number = 20
): UseAssetsProps {
  const [assets, setAssets] = useState<MediaLibrary.Asset[]>([]);
  const [loading, setLoading] = useState<boolean>(true);
  const [hasNextPage, setHasNextPage] = useState<boolean>(true);
  const [error, setError] = useState<Error | null>(null);
  const [endCursor, setEndCursor] = useState<string | undefined>(undefined);

  const fetchAssets = async (after?: string) => {
    try {
      setLoading(true);
      const { status } = await MediaLibrary.requestPermissionsAsync();
      
      if (status !== 'granted') {
        throw new Error('Permission to access media library denied');
      }

      const result = await MediaLibrary.getAssetsAsync({
        mediaType,
        first: limit,
        sortBy: [MediaLibrary.SortBy.creationTime],
        after,
      });

      setHasNextPage(result.hasNextPage);
      setEndCursor(result.endCursor);
      
      return result.assets;
    } catch (err) {
      setError(err instanceof Error ? err : new Error(String(err)));
      return [];
    } finally {
      setLoading(false);
    }
  };

  const loadMore = async () => {
    if (!hasNextPage || loading) return;
    
    const newAssets = await fetchAssets(endCursor);
    setAssets([...assets, ...newAssets]);
  };

  const refreshAssets = async () => {
    setEndCursor(undefined);
    const newAssets = await fetchAssets();
    setAssets(newAssets);
  };

  useEffect(() => {
    refreshAssets();
  }, [mediaType]);

  return { assets, loading, hasNextPage, loadMore, refreshAssets, error };
}
```

```typescript
import { View, StyleSheet, SafeAreaView, Alert } from 'react-native';
import { StackNavigationProp } from '@react-navigation/stack';
import { RouteProp } from '@react-navigation/native';
import { RootStackParamList } from '../../navigation/AppNavigator';
import { useAssets } from '../../hooks/useAssets';
import * as MediaLibrary from 'expo-media-library';
import CameraRollHeader from '../../components/cameraRoll/CameraRollHeader';
import CameraRollGrid from '../../components/cameraRoll/CameraRollGrid';
import ConfirmButton from '../../components/common/ConfirmButton';
import LoadingIndicator from '../../components/common/LoadingIndicator';

type CameraRollMainProps = {
  navigation: StackNavigationProp<RootStackParamList, 'CameraRollMain'>;
  route: RouteProp<RootStackParamList, 'CameraRollMain'>;
};

export default function CameraRollMain({ navigation, route }: CameraRollMainProps) {
  const { type = 1, btnText = '确定', maxSelect = 1, callback } = route.params || {};
  const [selectedItems, setSelectedItems] = useState<MediaLibrary.Asset[]>([]);
  const { assets, loading, hasNextPage, loadMore, refreshAssets, error } = useAssets(
    type === 1 ? MediaLibrary.MediaType.photo : MediaLibrary.MediaType.video
  );

  const handleSelectImage = useCallback((item: MediaLibrary.Asset) => {
    setSelectedItems(prev => {
      // 检查是否已被选中
      const alreadySelected = prev.some(selected => selected.id === item.id);
      
      if (alreadySelected) {
        // 如果已选中，则移除
        return prev.filter(selected => selected.id !== item.id);
      } else {
        // 如果未选中，且未超过最大选择数，则添加
        if (prev.length < maxSelect) {
          return [...prev, item];
        } else if (maxSelect === 1) {
          // 如果最大选择数为1，则替换唯一选择
          return [item];
        } else {
          // 如果已达到最大选择数，提示用户
          Alert.alert('提示', `最多只能选择 ${maxSelect} 张图片`);
          return prev;
        }
      }
    });
  }, [maxSelect]);

  const handleItemPress = useCallback((item: MediaLibrary.Asset) => {
    if (maxSelect === 1) {
      // 单选模式，直接设置为选中
      setSelectedItems([item]);
      // 如果是单图选择器回调模式，直接返回结果
      if (callback) {
        callback([item]);
        navigation.goBack();
      }
    } else {
      // 多选模式，切换选中状态
      handleSelectImage(item);
    }
  }, [maxSelect, callback, navigation, handleSelectImage]);

  const handleViewFullScreen = useCallback((item: MediaLibrary.Asset) => {
    navigation.navigate('CameraRollFullScreen', { uuid: item.id });
  }, [navigation]);

  const handleConfirm = useCallback(() => {
    if (callback) {
      callback(selectedItems);
      navigation.goBack();
    } else {
      navigation.navigate('EditorHome', { 
        imageUri: selectedItems.map(item => item.uri) 
      });
    }
  }, [callback, navigation, selectedItems]);

  if (error) {
    return (
      <SafeAreaView style={styles.container}>
        <CameraRollHeader title="选择照片" onClose={() => navigation.goBack()} />
        <View style={styles.errorContainer}>
          <Text style={styles.errorText}>无法加载相册: {error.message}</Text>
          <Button title="重试" onPress={refreshAssets} />
        </View>
      </SafeAreaView>
    );
  }

  return (
    <SafeAreaView style={styles.container}>
      <CameraRollHeader title="选择照片" onClose={() => navigation.goBack()} />
      
      <LoadingIndicator loading={loading && assets.length === 0} text="加载中...">
        <CameraRollGrid 
          assets={assets}
          selectedItems={selectedItems}
          onSelectItem={handleSelectImage}
          onItemPress={handleItemPress}
          onLoadMore={loadMore}
          loading={loading && assets.length > 0}
          maxSelect={maxSelect}
        />
      </LoadingIndicator>
      
      {selectedItems.length > 0 && (
        <ConfirmButton 
          text={`${btnText} (${selectedItems.length}/${maxSelect})`}
          onPress={handleConfirm}
        />
      )}
    </SafeAreaView>
  );
}
```

```typescript
// src/utils/ImageCacheUtil.ts
import * as FileSystem from 'expo-file-system';
import * as Crypto from 'expo-crypto';

interface CachedImage {
  uri: string;
  expiresAt: number;
}

const CACHE_FOLDER = `${FileSystem.cacheDirectory}image_cache/`;
const CACHE_EXPIRATION = 7 * 24 * 60 * 60 * 1000; // 7 days in milliseconds

class ImageCacheUtil {
  private cacheMap: Map<string, CachedImage> = new Map();
  private initialized = false;

  constructor() {
    this.init();
  }

  private async init() {
    try {
      // 确保缓存目录存在
      const { exists } = await FileSystem.getInfoAsync(CACHE_FOLDER);
      if (!exists) {
        await FileSystem.makeDirectoryAsync(CACHE_FOLDER, { intermediates: true });
      }

      // 读取缓存映射文件
      await this.loadCacheMap();
      this.initialized = true;
      
      // 清理过期缓存
      this.cleanExpiredCache();
    } catch (error) {
      console.error('Failed to initialize image cache:', error);
    }
  }

  private async loadCacheMap() {
    try {
      const mapFilePath = `${CACHE_FOLDER}cache_map.json`;
      const { exists } = await FileSystem.getInfoAsync(mapFilePath);
      
      if (exists) {
        const mapContent = await FileSystem.readAsStringAsync(mapFilePath);
        const map = JSON.parse(mapContent);
        
        // 转换为Map对象
        this.cacheMap = new Map(Object.entries(map));
      }
    } catch (error) {
      console.error('Failed to load cache map:', error);
      this.cacheMap = new Map();
    }
  }

  private async saveCacheMap() {
    try {
      const mapObject = Object.fromEntries(this.cacheMap);
      const mapContent = JSON.stringify(mapObject);
      const mapFilePath = `${CACHE_FOLDER}cache_map.json`;
      
      await FileSystem.writeAsStringAsync(mapFilePath, mapContent);
    } catch (error) {
      console.error('Failed to save cache map:', error);
    }
  }

  private async cleanExpiredCache() {
    const now = Date.now();
    const expiredKeys: string[] = [];
    
    // 找出过期的缓存项
    this.cacheMap.forEach((item, key) => {
      if (item.expiresAt < now) {
        expiredKeys.push(key);
      }
    });
    
    // 删除过期的缓存文件
    for (const key of expiredKeys) {
      const cachedItem = this.cacheMap.get(key);
      if (cachedItem) {
        try {
          await FileSystem.deleteAsync(cachedItem.uri, { idempotent: true });
          this.cacheMap.delete(key);
        } catch (error) {
          console.error(`Failed to delete expired cache for ${key}:`, error);
        }
      }
    }
    
    // 保存更新后的缓存映射
    if (expiredKeys.length > 0) {
      await this.saveCacheMap();
    }
  }

  private async getFileNameFromUrl(url: string): Promise<string> {
    const hash = await Crypto.digestStringAsync(
      Crypto.CryptoDigestAlgorithm.SHA256,
      url
    );
    return hash;
  }

  /**
   * 获取指定URL的缓存图像
   * @param url 图像URL
   * @returns 缓存的本地URI或原始URL
   */
  async getCachedImageUri(url: string): Promise<string> {
    // 等待初始化完成
    if (!this.initialized) {
      await new Promise<void>(resolve => {
        const checkInitialized = () => {
          if (this.initialized) {
            resolve();
          } else {
            setTimeout(checkInitialized, 50);
          }
        };
        checkInitialized();
      });
    }

    // 检查是否已缓存且未过期
    const cachedItem = this.cacheMap.get(url);
    if (cachedItem && cachedItem.expiresAt > Date.now()) {
      // 检查文件是否存在
      const { exists } = await FileSystem.getInfoAsync(cachedItem.uri);
      if (exists) {
        return cachedItem.uri;
      }
    }

    // 如果未缓存或已过期或文件不存在，则下载并缓存
    try {
      const fileName = await this.getFileNameFromUrl(url);
      const filePath = `${CACHE_FOLDER}${fileName}`;
      
      // 下载文件
      await FileSystem.downloadAsync(url, filePath);
      
      // 更新缓存映射
      const expiresAt = Date.now() + CACHE_EXPIRATION;
      this.cacheMap.set(url, { uri: filePath, expiresAt });
      
      // 保存缓存映射
      await this.saveCacheMap();
      
      return filePath;
    } catch (error) {
      console.error(`Failed to cache image from ${url}:`, error);
      return url; // 出错时返回原始URL
    }
  }

  /**
   * 清空所有缓存
   */
  async clearCache(): Promise<void> {
    try {
      // 删除缓存目录
      await FileSystem.deleteAsync(CACHE_FOLDER, { idempotent: true });
      
      // 重新创建缓存目录
      await FileSystem.makeDirectoryAsync(CACHE_FOLDER, { intermediates: true });
      
      // 清空缓存映射
      this.cacheMap.clear();
      await this.saveCacheMap();
    } catch (error) {
      console.error('Failed to clear image cache:', error);
    }
  }
}

// 导出单例
export default new ImageCacheUtil();
```

### 相册缩略图优化组件

```typescript
// src/components/cameraRoll/OptimizedImage.tsx
import React, { useState, useEffect } from 'react';
import { Image, StyleSheet, View, ActivityIndicator } from 'react-native';
import * as MediaLibrary from 'expo-media-library';
import * as FileSystem from 'expo-file-system';
import ImageCacheUtil from '../../utils/ImageCacheUtil';

interface OptimizedImageProps {
  asset: MediaLibrary.Asset;
  style?: any;
  optimizeForSize?: { width: number; height: number };
}

export default function OptimizedImage({
  asset,
  style,
  optimizeForSize
}: OptimizedImageProps) {
  const [uri, setUri] = useState<string | null>(null);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    let isMounted = true;

    const loadOptimizedImage = async () => {
      try {
        setLoading(true);
        
        // 构建缓存键
        const cacheKey = `${asset.id}_thumbnail_${optimizeForSize?.width || 'full'}_${optimizeForSize?.height || 'full'}`;
        
        // 尝试从缓存加载
        const cachedUri = await ImageCacheUtil.getCachedImageUri(cacheKey);
        
        if (cachedUri && cachedUri !== cacheKey) {
          // 已在缓存中找到
          if (isMounted) {
            setUri(cachedUri);
            setLoading(false);
          }
          return;
        }
        
        // 如果指定了尺寸优化，获取缩略图
        if (optimizeForSize) {
          const thumbnailUri = await MediaLibrary.getThumbnailAsync(asset, {
            width: optimizeForSize.width,
            height: optimizeForSize.height,
            quality: 0.7
          });
          
          if (isMounted) {
            setUri(thumbnailUri.uri);
            setLoading(false);
          }
          
          // 将缩略图缓存起来
          // 这里实际上没有下载逻辑，只是将路径映射保存起来
          await FileSystem.copyAsync({
            from: thumbnailUri.uri,
            to: `${FileSystem.cacheDirectory}${cacheKey}`
          }).catch(err => console.log('Cache thumbnail error:', err));
        } else {
          // 没有指定尺寸，直接使用原图
          if (isMounted) {
            setUri(asset.uri);
            setLoading(false);
          }
        }
      } catch (error) {
        console.error('Failed to load optimized image:', error);
        if (isMounted) {
          setUri(asset.uri);
          setLoading(false);
        }
      }
    };

    loadOptimizedImage();

    return () => {
      isMounted = false;
    };
  }, [asset, optimizeForSize]);

  if (loading) {
    return (
      <View style={[styles.container, style]}>
        <ActivityIndicator size="small" color="#3498db" />
      </View>
    );
  }

  return uri ? (
    <Image
      source={{ uri }}
      style={style}
      resizeMode="cover"
    />
  ) : null;
}

const styles = StyleSheet.create({
  container: {
    justifyContent: 'center',
    alignItems: 'center',
    backgroundColor: '#f9f9f9',
  },
});
```

以上组件和工具实现了相册模块的核心功能，包括图片的浏览、选择、预览和缓存。这些实现考虑了以下几点：

1. **性能优化**：
   - 图片缓存机制避免重复下载
   - 列表渲染优化（使用memo、优化渲染数量等）
   - 缩略图而非原图用于网格显示

2. **用户体验**：
   - 加载状态提示
   - 平滑的过渡动画
   - 直观的图片选择状态

3. **灵活性**：
   - 支持单选和多选模式
   - 提供回调机制用于不同的使用场景
